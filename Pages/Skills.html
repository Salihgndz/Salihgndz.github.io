import torch
import torch.nn as nn
import cv2
import numpy as np

class AutonomousNavigation(nn.Module):
    def __init__(self):
        super(AutonomousNavigation, self).__init__()
        self.conv1 = nn.Conv2d(3, 64, kernel_size=3, stride=1)
        self.relu = nn.ReLU()
        self.fc1 = nn.Linear(64 * 224 * 224, 128)

    def forward(self, x):
        x = self.conv1(x)
        x = self.relu(x)
        return self.fc1(x)

# Start realtime processing
model = AutonomousNavigation().cuda()
camera = cv2.VideoCapture(0)

while True:
    ret, frame = camera.read()
    if not ret: break

    tensor_frame = torch.from_numpy(frame).cuda()
    prediction = model(tensor_frame)

    # Send signals to MCU via UART
    serial.write(prediction.byte())

    if cv2.waitKey(1) & 0xFF == ord('q'):
        break
